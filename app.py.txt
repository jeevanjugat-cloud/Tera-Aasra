import streamlit as st
import sqlite3
import pandas as pd
from fpdf import FPDF
import urllib.parse
from datetime import datetime
import os

# --- NGO DETAILS ---
NGO_NAME = "Shabad Kirtan Naam Simran Satsang Sabha"
NGO_ADDRESS = "Registered, CB Tower, GT Road, Amritsar"

# --- DATABASE SETUP ---
def init_db():
    conn = sqlite3.connect('ngo_database.db')
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS donations
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, phone TEXT, amount REAL, date TEXT)''')
    c.execute('''CREATE TABLE IF NOT EXISTS expenses
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, description TEXT, amount REAL, date TEXT)''')
    conn.commit()
    conn.close()

init_db()

# --- PDF RECEIPT GENERATOR ---
class PDF(FPDF):
    def header(self):
        self.set_font('Arial', 'B', 14)
        self.cell(0, 10, NGO_NAME, 0, 1, 'C')
        self.set_font('Arial', '', 10)
        self.cell(0, 10, NGO_ADDRESS, 0, 1, 'C')
        self.line(10, 30, 200, 30)
        self.ln(10)

def generate_receipt(receipt_no, name, amount, date):
    pdf = PDF()
    pdf.add_page()
    pdf.set_font("Arial", size=12)
    
    pdf.cell(200, 10, txt="DONATION RECEIPT", ln=True, align='C')
    pdf.ln(10)
    
    pdf.cell(200, 10, txt=f"Receipt No: {receipt_no}", ln=True)
    pdf.cell(200, 10, txt=f"Date: {date}", ln=True)
    pdf.cell(200, 10, txt=f"Received with thanks from: {name}", ln=True)
    pdf.cell(200, 10, txt=f"Amount: Rs. {amount}/-", ln=True)
    pdf.ln(20)
    
    pdf.cell(200, 10, txt="Authorized Signatory", ln=True, align='R')
    
    filename = f"Receipt_{receipt_no}_{name.replace(' ', '_')}.pdf"
    pdf.output(filename)
    return filename

# --- APP INTERFACE ---
st.set_page_config(page_title="NGO Manager", layout="wide")
st.title(f"🙏 {NGO_NAME}")
st.write(NGO_ADDRESS)

tab1, tab2, tab3 = st.tabs(["Add Donation & Receipt", "Add Expenditure", "Account Summary"])

# TAB 1: DONATIONS
with tab1:
    st.header("New Donation")
    
    # Initialize session state flags for the conditional block outside the form
    if "receipt_ready" not in st.session_state:
        st.session_state.receipt_ready = False
        st.session_state.pdf_file = ""
        st.session_state.whatsapp_url = ""
        st.session_state.receipt_id = None

    with st.form("donation_form"):
        donor_name = st.text_input("Donor Name")
        donor_phone = st.text_input("Phone Number (with country code, e.g., 919876543210)")
        amount = st.number_input("Donation Amount (Rs.)", min_value=1)
        submit = st.form_submit_button("Save & Generate Receipt")

        if submit:
            if donor_name and amount:
                # Save to DB
                conn = sqlite3.connect('ngo_database.db')
                c = conn.cursor()
                current_date = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                c.execute("INSERT INTO donations (name, phone, amount, date) VALUES (?, ?, ?, ?)", 
                          (donor_name, donor_phone, amount, current_date))
                conn.commit()
                receipt_id = c.lastrowid
                conn.close()

                # Generate PDF file
                pdf_file = generate_receipt(receipt_id, donor_name, amount, current_date)
                
                # Update Session State flags to handle widgets outside the form
                st.session_state.receipt_ready = True
                st.session_state.pdf_file = pdf_file
                st.session_state.receipt_id = receipt_id
                
                # Format WhatsApp URL
                if donor_phone:
                    message = f"Waheguru Ji Ka Khalsa, Waheguru Ji Ki Fateh.\n\nDear {donor_name} Ji,\nThank you for your generous donation of Rs. {amount}/- to {NGO_NAME}. Your support is deeply appreciated."
                    encoded_message = urllib.parse.quote(message)
                    st.session_state.whatsapp_url = f"https://wa.me/{donor_phone}?text={encoded_message}"
                else:
                    st.session_state.whatsapp_url = ""
            else:
                st.error("Please enter donor name and amount.")

    # --- CONDITIONAL DISPLAY REGION (OUTSIDE THE FORM BLOCK) ---
    if st.session_state.receipt_ready:
        st.success(f"Donation saved! Receipt #{st.session_state.receipt_id} generated.")
        
        # Safe reading & execution of the generated PDF file outside form blocks
        if os.path.exists(st.session_state.pdf_file):
            with open(st.session_state.pdf_file, "rb") as file:
                st.download_button(
                    label="📄 Download PDF Receipt", 
                    data=file, 
                    file_name=st.session_state.pdf_file, 
                    mime="application/pdf"
                )
        
        # Display the WhatsApp contact mechanism
        if st.session_state.whatsapp_url:
            st.markdown(f"[💬 Click here to send WhatsApp Message (Attach the downloaded PDF manually)]({st.session_state.whatsapp_url})", unsafe_allow_html=True)

# TAB 2: EXPENDITURES
with tab2:
    st.header("Log Expenditure")
    with st.form("expense_form"):
        desc = st.text_input("Expense Description (e.g., Tent, Langar items, Electricity)")
        exp_amount = st.number_input("Expense Amount (Rs.)", min_value=1)
        exp_submit = st.form_submit_button("Save Expense")

        if exp_submit:
            if desc and exp_amount:
                conn = sqlite3.connect('ngo_database.db')
                c = conn.cursor()
                current_date = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                c.execute("INSERT INTO expenses (description, amount, date) VALUES (?, ?, ?)", 
                          (desc, exp_amount, current_date))
                conn.commit()
                conn.close()
                st.success("Expenditure saved successfully!")
            else:
                st.error("Please enter description and amount.")

# TAB 3: ACCOUNT SUMMARY
with tab3:
    st.header("Financial Overview")
    conn = sqlite3.connect('ngo_database.db')
    
    donations_df = pd.read_sql_query("SELECT * FROM donations", conn)
    expenses_df = pd.read_sql_query("SELECT * FROM expenses", conn)
    conn.close()
    
    total_donations = donations_df['amount'].sum() if not donations_df.empty else 0
    total_expenses = expenses_df['amount'].sum() if not expenses_df.empty else 0
    balance = total_donations - total_expenses
    
    col1, col2, col3 = st.columns(3)
    col1.metric("Total Donations", f"Rs. {total_donations:,.2f}")
    col2.metric("Total Expenses", f"Rs. {total_expenses:,.2f}")
    col3.metric("Current Balance", f"Rs. {balance:,.2f}")
    
    st.subheader("Recent Donations")
    st.dataframe(donations_df.tail(10))
    
    st.subheader("Recent Expenses")
    st.dataframe(expenses_df.tail(10))
